#include "debug.h"
#include <stdint.h>

/*
 * CH32V003J4M6
 *
 * PC1 = I2C SDA
 * PC2 = I2C SCL
 *
 * OLED = SSD1306 64x32
 */


/* ============================================================
   I2C
   ============================================================ */

#define OLED_ADDRESS_1     0x3C
#define OLED_ADDRESS_2     0x3D


static void i2c_init(void)
{
    GPIO_InitTypeDef GPIO_InitStructure = {0};
    I2C_InitTypeDef I2C_InitStructure = {0};


    RCC_APB2PeriphClockCmd(
        RCC_APB2Periph_GPIOC |
        RCC_APB2Periph_AFIO,
        ENABLE
    );


    RCC_APB1PeriphClockCmd(
        RCC_APB1Periph_I2C1,
        ENABLE
    );


    /*
     * PC1 = SDA
     * PC2 = SCL
     */

    GPIO_InitStructure.GPIO_Pin =
        GPIO_Pin_1 |
        GPIO_Pin_2;

    GPIO_InitStructure.GPIO_Mode =
        GPIO_Mode_AF_OD;

    GPIO_InitStructure.GPIO_Speed =
        GPIO_Speed_10MHz;

    GPIO_Init(
        GPIOC,
        &GPIO_InitStructure
    );


    /*
     * Start conservatively at 100 kHz.
     */

    I2C_InitStructure.I2C_ClockSpeed =
        100000;

    I2C_InitStructure.I2C_Mode =
        I2C_Mode_I2C;

    I2C_InitStructure.I2C_DutyCycle =
        I2C_DutyCycle_2;

    I2C_InitStructure.I2C_OwnAddress1 =
        0x00;

    I2C_InitStructure.I2C_Ack =
        I2C_Ack_Enable;

    I2C_InitStructure.I2C_AcknowledgedAddress =
        I2C_AcknowledgedAddress_7bit;


    I2C_Init(
        I2C1,
        &I2C_InitStructure
    );


    I2C_Cmd(
        I2C1,
        ENABLE
    );
}


/* ============================================================
   I2C DEVICE TEST
   ============================================================ */

static uint8_t i2c_probe(uint8_t address)
{
    uint32_t timeout;


    /*
     * Make sure the bus isn't busy.
     */

    timeout = 100000;

    while (
        I2C_GetFlagStatus(
            I2C1,
            I2C_FLAG_BUSY
        )
    )
    {
        if (--timeout == 0)
            return 0;
    }


    /*
     * START
     */

    I2C_GenerateSTART(
        I2C1,
        ENABLE
    );


    timeout = 100000;

    while (
        !I2C_CheckEvent(
            I2C1,
            I2C_EVENT_MASTER_MODE_SELECT
        )
    )
    {
        if (--timeout == 0)
            return 0;
    }


    /*
     * Send address.
     */

    I2C_Send7bitAddress(
        I2C1,
        address << 1,
        I2C_Direction_Transmitter
    );


    /*
     * Wait for ACK.
     */

    timeout = 100000;

    while (timeout--)
    {
        if (
            I2C_CheckEvent(
                I2C1,
                I2C_EVENT_MASTER_TRANSMITTER_MODE_SELECTED
            )
        )
        {
            I2C_GenerateSTOP(
                I2C1,
                ENABLE
            );

            return 1;
        }


        /*
         * Check for NACK.
         */

        if (
            I2C_GetFlagStatus(
                I2C1,
                I2C_FLAG_AF
            )
        )
        {
            I2C_ClearFlag(
                I2C1,
                I2C_FLAG_AF
            );

            I2C_GenerateSTOP(
                I2C1,
                ENABLE
            );

            return 0;
        }
    }


    I2C_GenerateSTOP(
        I2C1,
        ENABLE
    );

    return 0;
}


/* ============================================================
   OLED LOW LEVEL
   ============================================================ */

static void oled_command(
    uint8_t address,
    uint8_t command
)
{
    uint8_t data[2];

    data[0] = 0x00;
    data[1] = command;


    uint32_t timeout;


    while (
        I2C_GetFlagStatus(
            I2C1,
            I2C_FLAG_BUSY
        )
    );


    /*
     * START
     */

    I2C_GenerateSTART(
        I2C1,
        ENABLE
    );

    timeout = 100000;

    while (
        !I2C_CheckEvent(
            I2C1,
            I2C_EVENT_MASTER_MODE_SELECT
        )
    )
    {
        if (--timeout == 0)
            return;
    }


    /*
     * Address
     */

    I2C_Send7bitAddress(
        I2C1,
        address << 1,
        I2C_Direction_Transmitter
    );


    timeout = 100000;

    while (
        !I2C_CheckEvent(
            I2C1,
            I2C_EVENT_MASTER_TRANSMITTER_MODE_SELECTED
        )
    )
    {
        if (--timeout == 0)
        {
            I2C_GenerateSTOP(
                I2C1,
                ENABLE
            );

            return;
        }
    }


    /*
     * Control byte
     */

    I2C_SendData(
        I2C1,
        data[0]
    );


    timeout = 100000;

    while (
        !I2C_CheckEvent(
            I2C1,
            I2C_EVENT_MASTER_BYTE_TRANSMITTED
        )
    )
    {
        if (--timeout == 0)
            return;
    }


    /*
     * Command
     */

    I2C_SendData(
        I2C1,
        data[1]
    );


    timeout = 100000;

    while (
        !I2C_CheckEvent(
            I2C1,
            I2C_EVENT_MASTER_BYTE_TRANSMITTED
        )
    )
    {
        if (--timeout == 0)
            return;
    }


    I2C_GenerateSTOP(
        I2C1,
        ENABLE
    );
}


/* ============================================================
   OLED INITIALIZATION
   ============================================================ */

static void oled_init(
    uint8_t address
)
{
    Delay_Ms(100);


    oled_command(address, 0xAE);

    oled_command(address, 0xD5);
    oled_command(address, 0x80);

    oled_command(address, 0xA8);
    oled_command(address, 0x1F);

    oled_command(address, 0xD3);
    oled_command(address, 0x00);

    oled_command(address, 0x40);

    oled_command(address, 0x8D);
    oled_command(address, 0x14);

    oled_command(address, 0x20);
    oled_command(address, 0x00);

    oled_command(address, 0xA1);

    oled_command(address, 0xC8);

    oled_command(address, 0xDA);
    oled_command(address, 0x02);

    oled_command(address, 0x81);
    oled_command(address, 0x8F);

    oled_command(address, 0xD9);
    oled_command(address, 0xF1);

    oled_command(address, 0xDB);
    oled_command(address, 0x40);

    oled_command(address, 0xA4);

    oled_command(address, 0xA6);

    oled_command(address, 0xAF);
}


/* ============================================================
   SEND OLED DATA
   ============================================================ */

static void oled_write_data(
    uint8_t address,
    const uint8_t *data,
    uint16_t length
)
{
    uint32_t timeout;


    while (
        I2C_GetFlagStatus(
            I2C1,
            I2C_FLAG_BUSY
        )
    );


    I2C_GenerateSTART(
        I2C1,
        ENABLE
    );


    timeout = 100000;

    while (
        !I2C_CheckEvent(
            I2C1,
            I2C_EVENT_MASTER_MODE_SELECT
        )
    )
    {
        if (--timeout == 0)
            return;
    }


    I2C_Send7bitAddress(
        I2C1,
        address << 1,
        I2C_Direction_Transmitter
    );


    timeout = 100000;

    while (
        !I2C_CheckEvent(
            I2C1,
            I2C_EVENT_MASTER_TRANSMITTER_MODE_SELECTED
        )
    )
    {
        if (--timeout == 0)
            return;
    }


    /*
     * Data control byte.
     */

    I2C_SendData(
        I2C1,
        0x40
    );


    timeout = 100000;

    while (
        !I2C_CheckEvent(
            I2C1,
            I2C_EVENT_MASTER_BYTE_TRANSMITTED
        )
    )
    {
        if (--timeout == 0)
            return;
    }


    /*
     * Display RAM.
     */

    for (
        uint16_t i = 0;
        i < length;
        i++
    )
    {
        I2C_SendData(
            I2C1,
            data[i]
        );


        timeout = 100000;

        while (
            !I2C_CheckEvent(
                I2C1,
                I2C_EVENT_MASTER_BYTE_TRANSMITTED
            )
        )
        {
            if (--timeout == 0)
                return;
        }
    }


    I2C_GenerateSTOP(
        I2C1,
        ENABLE
    );
}


/* ============================================================
   SHOW SOLID SCREEN
   ============================================================ */

static void oled_fill(
    uint8_t address,
    uint8_t pattern
)
{
    /*
     * 64 x 32 = 256 bytes.
     */

    uint8_t buffer[64];


    /*
     * Set horizontal addressing.
     */

    oled_command(address, 0x21);
    oled_command(address, 0x00);
    oled_command(address, 0x3F);

    oled_command(address, 0x22);
    oled_command(address, 0x00);
    oled_command(address, 0x03);


    /*
     * Fill each page.
     */

    for (
        uint8_t page = 0;
        page < 4;
        page++
    )
    {
        for (
            uint8_t i = 0;
            i < 64;
            i++
        )
        {
            buffer[i] = pattern;
        }


        oled_write_data(
            address,
            buffer,
            64
        );
    }
}


/* ============================================================
   MAIN
   ============================================================ */

int main(void)
{
    uint8_t oledAddress = 0;


    SystemCoreClockUpdate();

    Delay_Init();

    i2c_init();

    /*
     * Give the OLED time to power up.
     */

    Delay_Ms(500);


    /*
     * Search for 0x3C.
     */

    if (
        i2c_probe(0x3C)
    )
    {
        oledAddress = 0x3C;
    }


    /*
     * If 0x3C wasn't found,
     * try 0x3D.
     */

    if (
        oledAddress == 0 &&
        i2c_probe(0x3D)
    )
    {
        oledAddress = 0x3D;
    }


    /*
     * If an OLED was found,
     * initialize it and turn the
     * whole display ON.
     */

    if (oledAddress != 0)
    {
        oled_init(
            oledAddress
        );


        /*
         * Solid white screen.
         *
         * This proves:
         *
         * CH32V003
         *   ↓
         * I2C
         *   ↓
         * OLED
         *
         * is working.
         */

        oled_fill(
            oledAddress,
            0xFF
        );
    }


    while (1)
    {
        /*
         * Nothing else to do.
         *
         * The screen remains solid white.
         */

        Delay_Ms(1000);
    }
}
